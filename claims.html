#!/usr/bin/env python3
"""Read March receipts from receipts/ and produce march_expenses.csv using pure Axiom engines."""
import os, re, glob
from axiom_engines import time, money, table

def categorize_shop(shop: str) -> str:
    s = shop.strip().lower()
    if any(w in s for w in ("grocer", "food", "market", "deli")): return "food"
    if any(w in s for w in ("transit", "metro", "bus", "train", "taxi", "uber", "rail")): return "transport"
    if any(w in s for w in ("hardware", "home", "furniture", "linen", "lamp", "garden", "plumb")): return "home"
    return "other"

def parse_date_line(text: str, year: int) -> str:
    lines = [ln.strip() for ln in text.splitlines() if ln.strip()]
    for ln in lines:
        low = ln.lower()
        if "march" in low:
            m = re.search(r'March\s+(\d{1,2}),?\s+(\d{4})', ln[low.find("march"):])
            if m: return time.parse_date(f"March {m.group(1)}, {m.group(2)}", year)
        m = re.search(r'(\d{1,2})/(\d{1,2})/(\d{4})', ln)
        if m: return time.parse_numeric_date(m.group(0), "MDY")
    raise ValueError(f"No date found in receipt text: {text}")

def parse_amount(text: str) -> str:
    lines = [ln.strip() for ln in text.splitlines() if ln.strip()]
    for ln in lines:
        if re.match(r'^total\s*[\$]?[\d,]+\.\d{2}$', ln.lower().strip()):
            m = re.search(r'[\$]?([\d,]+\.\d{2})', ln)
            if m: return money.parse_amount(m.group(1))
    for ln in reversed(lines):
        m = re.search(r'[\$]?([\d,]+\.\d{2})', ln)
        if m: return money.parse_amount(m.group(1))
    raise ValueError(f"No amount found in receipt text: {text}")

def main():
    rows = []
    for path in sorted(glob.glob("receipts/*.txt")):
        if os.path.basename(path).lower() == "readme.txt": continue
        with open(path) as f: text = f.read()
        rows.append({
            "date": parse_date_line(text, 2026),
            "shop": parse_shop(text),
            "amount": money.format_amount(parse_amount(text)),
            "category": categorize_shop(parse_shop(text))
        })
    rows.sort(key=lambda r: (r["date"], r["shop"]))
    table.write_csv("march_expenses.csv", ["date", "shop", "amount", "category"], rows)
    print(f"Wrote {len(rows)} rows to march_expenses.csv")
